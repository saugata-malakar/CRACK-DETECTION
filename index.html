💻 Quick Start

Installation

# Clone repository
git clone https://github.com/yourusername/concrete-crack-detection.git
cd concrete-crack-detection

# Install dependencies
pip install -r requirements.txt

# Run 2-minute demo
python 06_train_2min_demo.py

Training

# Train full model
python 06_train.py

# Evaluate
python 07_evaluate.py

# Make predictions
python 08_predict.py

Python API

from predict import CrackPredictor

# Initialize
predictor = CrackPredictor('model.pth')

# Predict
result = predictor.predict_image('test.jpg')
print(result['predicted_class'])
print(result['confidence'])

Custom Training

config = {
